ESP32-CAM Important Note 
The ESP32-CAM does NOT have a built-in USB port for programming. 
You need an FTDI USB-to-Serial adapter (CH340 or CP2102) to flash it. 
Connect: FTDI TX → CAM RX, FTDI RX → CAM TX, GND → GND, 5V → 5V. 
Hold GPIO0 to GND during upload, then release for normal operation. 
After flashing, the FTDI is removed — power via 5V from USB hub or power bank.

Part 2 — Circuit Diagrams — Every ESP32 Board Wired 
2.1 ESP32-CAM Node — Camera Device 
The ESP32-CAM acts as your highest-traffic device, simulating an IP camera. It does NOT need to stream real 
video — it sends realistic camera-class telemetry over MQTT. The onboard LED serves as the trust indicator. 
 
ESP32-CAM Pinout & Wiring 
ESP32-CAM Board Connections: 
 
  Power: 
    5V pin  → USB Hub / Power Bank 5V 
    GND pin → Common ground 
 
  Programming (FTDI adapter — REMOVE after flashing): 
    GPIO0   → GND  (hold during upload, release after) 
    U0TXD   → FTDI RX 
    U0RXD   → FTDI TX 
    GND     → FTDI GND 
    5V      → FTDI 5V 
 
  RGB LED (Trust Score Indicator): 
    GPIO 12 → 330-ohm resistor → Red LED anode 
    GPIO 13 → 330-ohm resistor → Green LED anode 
    GPIO 15 → 330-ohm resistor → Blue LED anode 
    All LED cathodes → GND 
 
  Note: GPIO 0, 2, 4 are used by camera — do NOT connect anything to these. 
  The onboard flash LED is on GPIO 4. 
 
  LED Colors = Trust Score: 
    GREEN  → Trust > 70 (normal) 
    YELLOW → Red+Green ON  → Trust 40-70 (suspicious) 
    RED    → Trust < 40 (critical / under attack) 
 
2.2 ESP32 + DHT11 Node — Temperature/Humidity Sensor Device 
ESP32 Sensor Node Wiring 
ESP32 Dev Board (38-pin) + DHT11 sensor: 
 
  Power: 
    3.3V pin → DHT11 pin 1 (VCC) 
    GND      → DHT11 pin 4 (GND) 
    Note: DHT11 works on 3.3V despite being rated 3-5V 
 
  DHT11 Data Connection: 
    GPIO 4  → DHT11 pin 2 (DATA) 
    10k-ohm pull-up: between GPIO 4 and 3.3V 
    (DHT11 pin 3 is NC — leave unconnected) 
 
  RGB LED (Trust Score Indicator): 
    GPIO 25 → 330-ohm → Red LED anode

2.3 ESP32 + PIR Motion Node — Motion Detector Device 
ESP32 PIR Node Wiring 
ESP32 Dev Board (38-pin) + HC-SR501 PIR sensor: 
 
  Power: 
    5V pin  → PIR VCC  (PIR needs 5V, NOT 3.3V) 
    GND     → PIR GND 
    Note: Use the ESP32 5V pin (Vin) which is USB voltage 
 
  PIR Output Connection: 
    PIR OUT → GPIO 34 (input only pin — perfect for PIR) 
    Note: PIR output is 3.3V logic, safe for ESP32 direct connection 
 
  PIR Adjustment Pots (on HC-SR501 board): 
    Left pot  → Sensitivity (turn clockwise for more sensitive) 
    Right pot → Delay time (set to minimum ~3 seconds for demo) 
 
  RGB LED: 
    GPIO 25 → 330-ohm → Red 
    GPIO 26 → 330-ohm → Green 
    GPIO 27 → 330-ohm → Blue 
    Cathodes → GND 
 
  Extra: Red alert LED 
    GPIO 32 → 330-ohm → Red LED → GND 
    Blinks on motion detection AND on attack alert

float g_trustScore = 100.0;   // Updated by server commands 
bool  g_quarantined = false; 
 
// RGB LED: show trust score visually 
void setTrustLED(float score) { 
  if (score >= 70.0) { 
    // GREEN = trusted 
    digitalWrite(LED_RED_PIN,   LOW); 
    digitalWrite(LED_GREEN_PIN, HIGH); 
    digitalWrite(LED_BLUE_PIN,  LOW); 
  } else if (score >= 40.0) { 
    // YELLOW = suspicious (red + green) 
    digitalWrite(LED_RED_PIN,   HIGH); 
    digitalWrite(LED_GREEN_PIN, HIGH); 
    digitalWrite(LED_BLUE_PIN,  LOW); 
  } else { 
    // RED = critical / under attack 
    digitalWrite(LED_RED_PIN,   HIGH); 
    digitalWrite(LED_GREEN_PIN, LOW); 
    digitalWrite(LED_BLUE_PIN,  LOW); 
  } 
} 
 
// Handle commands from AI server 
void mqttCallback(char* topic, byte* payload, unsigned int length) { 
  StaticJsonDocument<256> doc; 
  deserializeJson(doc, payload, length); 
 
  const char* action = doc["action"]; 
  float score = doc["trust_score"] | 100.0; 
 
  g_trustScore = score; 
  setTrustLED(score); 
 
  if (strcmp(action, "quarantine") == 0) { 
    g_quarantined = true; 
    // Flash red 3 times = quarantine alert 
    for (int i=0; i<3; i++) { 
      digitalWrite(LED_RED_PIN, HIGH); 
      delay(200); 
      digitalWrite(LED_RED_PIN, LOW);

void connectWiFi() { 
  Serial.print("Connecting to WiFi: "); Serial.println(WIFI_SSID); 
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD); 
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); } 
  Serial.print("\nConnected! IP: "); Serial.println(WiFi.localIP()); 
} 
 
void connectMQTT() { 
  mqttClient.setServer(MQTT_BROKER, MQTT_PORT); 
  mqttClient.setCallback(mqttCallback); 
  while (!mqttClient.connected()) { 
    Serial.print("Connecting to MQTT..."); 
    if (mqttClient.connect(DEVICE_ID)) { 
      Serial.println(" connected!"); 
      mqttClient.subscribe(TOPIC_COMMAND); 
      // Announce device online 
      StaticJsonDocument<128> status; 
      status["device_id"] = DEVICE_ID; 
      status["device_type"] = DEVICE_TYPE; 
      status["status"] = "online"; 
      char buf[128]; serializeJson(status, buf); 
      mqttClient.publish(TOPIC_STATUS, buf); 
    } else { delay(2000); } 
  } 
} 
 
void mqttLoop() { 
  if (!mqttClient.connected()) connectMQTT(); 
  mqttClient.loop(); 
}

7.1 FastAPI MQTT Consumer (main.py excerpt) 
The FastAPI backend subscribes to all MQTT telemetry, extracts 14D feature vectors, and routes to the ML 
pipeline. Every message from physical AND virtual devices follows the same path. 
 
# backend/main.py (core MQTT consumer section) 
import asyncio, json 
import asyncio_mqtt as aiomqtt 
from feature_extraction import extract_features 
from trust_engine import evaluate_device 
from db.redis_client import cache_trust_score 
from db.influx_client import write_telemetry 
from websocket_manager import broadcast_trust_update 
 
async def mqtt_consumer(): 
    async with aiomqtt.Client('localhost') as client: 
        await client.subscribe('devicedna/telemetry') 
        async for message in client.messages: 
            try: 
                raw = json.loads(message.payload) 
                device_id = raw['device_id'] 
                device_type = raw['device_type'] 
 
                # 1. Extract 14D feature vector 
                features = extract_features(raw) 
 
                # 2. Write raw telemetry to InfluxDB 
                await write_telemetry(device_id, raw, features) 
 
                # 3. Run ML pipeline 
                trust_result = await evaluate_device(device_id, device_type, features) 
 
                # 4. Cache trust score in Redis 
                await cache_trust_score(device_id, trust_result) 
 
                # 5. Push to dashboard via WebSocket 
                await broadcast_trust_update(device_id, trust_result) 
 
                # 6. Auto-quarantine if critical 
                if trust_result['trust_score'] < 30: 
                    await send_quarantine_command(device_id) 
 
            except Exception as e: 
                print(f'[ERROR] Processing {device_id}: {e}')

import numpy as np 
 
def extract_features(raw: dict) -> np.ndarray: 
    '''Convert raw MQTT payload to 14D feature vector.''' 
    features = np.array([ 
        float(raw.get('total_flows',     0)),       # dim 0 
        float(raw.get('total_bytes',     0)),       # dim 1 
        float(raw.get('total_packets',   0)),       # dim 2 
        float(raw.get('avg_packet_size', 0)),       # dim 3 
        float(raw.get('avg_duration_ms', 500)),     # dim 4 
        float(raw.get('tcp_ratio',       0.5)),     # dim 5 
        float(raw.get('udp_ratio',       0.1)),     # dim 6 
        float(raw.get('http_ratio',      0.05)),    # dim 7 
        float(raw.get('https_ratio',     0.2)),     # dim 8 
        float(raw.get('dns_ratio',       0.05)),    # dim 9
